### 备份脚本

```shell
#!/bin/bash
# simple-backup.sh

BACKUP_FILE="/home/$(whoami)/system-backup-$(date +%Y%m%d-%H%M).tar.gz"
EXCLUDE_FILE="/tmp/exclude-list.txt"

# 创建排除列表
cat > $EXCLUDE_FILE << EOF
/dev/*
/proc/*
/sys/*
/tmp/*
/run/*
/mnt/*
/media/*
/lost+found
/var/cache/pacman/pkg/*
/home/*/.cache/*
*/system-backup-*.tar.gz
EOF

echo "开始创建系统备份..."
echo "备份文件: $BACKUP_FILE"

# 创建备份（修正版）
sudo tar -czf "$BACKUP_FILE" \
    -X "$EXCLUDE_FILE" \
    --exclude=./var/log/* \
    /

# 检查备份是否成功
if [ $? -eq 0 ]; then
    echo "备份完成！"
    echo "文件大小: $(du -h "$BACKUP_FILE" | cut -f1)"
else
    echo "备份失败！"
    # 清理可能创建的损坏文件
    rm -f "$BACKUP_FILE"
fi

# 清理临时文件
rm -f "$EXCLUDE_FILE"
```

### 恢复脚本

```shell
#!/bin/bash
# simple-restore.sh

if [ $# -ne 1 ]; then
    echo "用法: $0 <备份文件.tar.gz>"
    exit 1
fi

BACKUP_FILE="$1"
TARGET_PARTITION="/dev/sda1"  # 根据实际情况修改

if [ ! -f "$BACKUP_FILE" ]; then
    echo "错误: 备份文件不存在"
    exit 1
fi

echo "警告：此操作将格式化 $TARGET_PARTITION"
read -p "确认恢复？(y/N): " confirm

if [[ ! $confirm =~ ^[Yy]$ ]]; then
    echo "操作已取消"
    exit 0
fi

# 格式化分区
sudo mkfs.ext4 -F $TARGET_PARTITION

# 挂载分区
sudo mkdir -p /mnt/restore
sudo mount $TARGET_PARTITION /mnt/restore

# 解压备份
echo "正在恢复系统..."
sudo tar -xzf $BACKUP_FILE -C /mnt/restore

# 生成fstab
sudo genfstab -U /mnt/restore > /mnt/restore/etc/fstab

# 安装引导程序
sudo arch-chroot /mnt/restore grub-install /dev/sda
sudo arch-chroot /mnt/restore grub-mkconfig -o /boot/grub/grub.cfg

# 卸载
sudo umount /mnt/restore

echo "系统恢复完成！"
